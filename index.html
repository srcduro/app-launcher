-- ============================================================================ -- resolve_login_email_with_contact.sql -- -- Purpose: -- The Employee-Code login path calls DUROPLY.resolve_login_email(p_code), -- which returns only the Supabase Auth "tagged" email (the address the -- OTP is actually mailed to — this can be a synthetic, per-account -- address under the multi-user-same-email scheme). The App Launcher's -- "Code sent to …" text should instead show the human-facing -- DUROPLY.user_profiles.CONTACT_EMAIL. -- -- This migration does NOT touch or replace resolve_login_email() — -- it adds a new, additive wrapper function that calls the existing one -- (zero risk to the current login flow) and also looks up contact_email -- for the same employee code, returning both as JSON. -- -- Assumption to verify before running: -- The employee-code column on DUROPLY.user_profiles is named -- "employee_id" (this matches the wording already used in the front-end -- code's own comments). If your actual column name differs, change the -- WHERE clause below accordingly before running. -- -- How to apply: -- 1. Open Supabase Dashboard → SQL Editor (or `psql`) on this project. -- 2. Run the CREATE OR REPLACE FUNCTION block below. -- 3. Run the GRANT statement so the anon key (used before login) can -- call it — same exposure level as the existing resolve_login_email(). -- 4. Test it directly in the SQL editor first, e.g.: -- select "DUROPLY".resolve_login_email_with_contact('S1234A'); -- It should return something like: -- {"auth_email": "s1234a@duroply.internal", "contact_email": "jdoe@duroply.com"} -- 5. Only after that looks right, deploy the updated App_Launcher HTML -- (already wired to call this new function) to replace the live one. -- -- Rollback: -- drop function if exists "DUROPLY".resolve_login_email_with_contact(text); -- (Also revert the App Launcher HTML to the previous version if you need -- to fully undo — it will start failing employee-code sign-in otherwise, -- since that HTML no longer calls the original resolve_login_email().) -- ============================================================================ create or replace function "DUROPLY".resolve_login_email_with_contact(p_code text) returns jsonb language plpgsql security definer set search_path = "DUROPLY", public as $$ declare v_auth_email text; v_contact_email text; begin -- Reuse the existing, already-tested employee-code → auth-email lookup -- untouched, so this wrapper can never regress current login behaviour. select "DUROPLY".resolve_login_email(p_code) into v_auth_email; if v_auth_email is null or length(trim(v_auth_email)) = 0 then return jsonb_build_object('auth_email', null, 'contact_email', null); end if; -- Best-effort enrichment only: if this lookup doesn't find a row (e.g. -- a column-name mismatch), we still return the auth_email above and the -- front end falls back to showing that instead of contact_email. select contact_email into v_contact_email from "DUROPLY".user_profiles where upper(employee_id) = upper(p_code) limit 1; return jsonb_build_object( 'auth_email', lower(trim(v_auth_email)), 'contact_email', case when v_contact_email is not null then lower(trim(v_contact_email)) else null end ); end; $$; -- Match whatever grants the existing resolve_login_email() has today -- (it must already be callable by 'anon' pre-login, or employee-code -- sign-in wouldn't work at all) — mirror that here. grant execute on function "DUROPLY".resolve_login_email_with_contact(text) to anon, authenticated;
